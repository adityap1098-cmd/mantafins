    Create `src/lib/auth.ts`:
    ```typescript
    import { prisma } from './prisma'

    export const SESSION_COOKIE_NAME = 'manta_session'
    export const SESSION_DURATION_MS = 7 * 24 * 60 * 60 * 1000 // 7 days

    export async function createSession(): Promise<string> {
      const { id: token } = await prisma.session.create({
        data: {
          token: crypto.randomUUID(),
          expiresAt: new Date(Date.now() + SESSION_DURATION_MS),
        },
      })
      // Return the token field, not the id
      const session = await prisma.session.findFirst({ orderBy: { createdAt: 'desc' } })
      return session!.token
    }

    export async function validateSession(token: string): Promise<boolean> {
      if (!token) return false
      const session = await prisma.session.findUnique({ where: { token } })
      if (!session) return false
      if (session.expiresAt < new Date()) {
        await prisma.session.delete({ where: { token } })
        return false
      }
      return true
    }

    export async function deleteSession(token: string): Promise<void> {
      await prisma.session.deleteMany({ where: { token } })
    }
    ```

    ```typescript
    import { NextRequest, NextResponse } from 'next/server'
    import { createSession, SESSION_COOKIE_NAME, SESSION_DURATION_MS } from '@/lib/auth'

    export async function POST(req: NextRequest) {
      const { password } = await req.json()
      if (!password || password !== process.env.APP_PASSWORD) {
        return NextResponse.json({ error: 'Password salah' }, { status: 401 })
      }
      const token = await createSession()
      const res = NextResponse.json({ success: true })
      res.cookies.set(SESSION_COOKIE_NAME, token, {
        httpOnly: true,
        secure: process.env.NODE_ENV === 'production',
        sameSite: 'lax',
        maxAge: SESSION_DURATION_MS / 1000, // seconds
        path: '/',
      })
      return res
    }
    ```

    ```typescript
    export async function GET(req: NextRequest) {
      const token = req.cookies.get(SESSION_COOKIE_NAME)?.value
      if (!token) return NextResponse.json({ authenticated: false }, { status: 401 })
      const valid = await validateSession(token)
    ```

    Create `middleware.ts` in the project root (manta-finance/middleware.ts — NOT inside src/):
    ```typescript
    import { NextRequest, NextResponse } from 'next/server'
    import { SESSION_COOKIE_NAME } from '@/lib/auth'

    // Routes that do NOT require authentication
    const PUBLIC_PATHS = ['/login', '/api/auth/login', '/api/auth/logout']

    export async function middleware(req: NextRequest) {
      const { pathname } = req.nextUrl

      // Allow public paths through
      if (PUBLIC_PATHS.some((p) => pathname.startsWith(p))) {
        return NextResponse.next()
      }

      // Allow static files and Next.js internals through
      if (
        pathname.startsWith('/_next') ||
        pathname.startsWith('/favicon') ||
        pathname.includes('.')
      ) {
        return NextResponse.next()
      }

      // Check session cookie presence (lightweight check in middleware)
      const token = req.cookies.get(SESSION_COOKIE_NAME)?.value
      if (!token) {
        return NextResponse.redirect(new URL('/login', req.url))
      }

      // Token exists — let the request through. Full DB validation happens in API routes.
      // Middleware runs on Edge runtime; DB calls are handled server-side in routes/pages.
      return NextResponse.next()
    }

    export const config = {
      matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
    }
    ```

    IMPORTANT: The middleware does a lightweight cookie presence check only (no DB call) because:
    1. Middleware runs on Edge runtime which has restrictions on Node.js APIs
    2. Full session validation (DB check) happens in individual API routes and server components
    3. Cookie-only check is sufficient for redirect logic — expired sessions are caught at the API level

    If Prisma cannot be imported in middleware due to Edge runtime restrictions, the SESSION_COOKIE_NAME constant can be inlined directly: `const SESSION_COOKIE_NAME = 'manta_session'`
