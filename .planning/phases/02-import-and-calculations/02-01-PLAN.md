model Period {
  id        String            @id @default(cuid())
  name      String
  month     Int
  year      Int
  createdAt DateTime          @default(now())
  products  ProductSnapshot[]
  sales     Sale[]
}

model ProductSnapshot {
  id        String  @id @default(cuid())
  periodId  String
  period    Period  @relation(fields: [periodId], references: [id])
  sku       String
  name      String
  category  String
  hpp       Float
  hargaJual Float
  stock     Int

  @@unique([periodId, sku])
}

model Sale {
  id         String     @id @default(cuid())
  periodId   String
  period     Period     @relation(fields: [periodId], references: [id])
  date       DateTime
  refNo      String
  supplier   String
  customer   String
  grandTotal Float
  paid       Float
  balance    Float
  status     String
  items      SaleItem[]
}

model SaleItem {
  id          String  @id @default(cuid())
  saleId      String
  sale        Sale    @relation(fields: [saleId], references: [id])
  sku         String
  productName String
  qty         Int
  hppUnit     Float
  hargaUnit   Float
  totalHpp    Float
  totalHarga  Float
}

model PaymentLog {
  id       String   @id @default(cuid())
  saleId   String
  refNo    String
  customer String
  amount   Float
  date     DateTime
  note     String?
}

Create manta-finance/lib/prisma.ts with a global PrismaClient singleton. Pattern must use @prisma/adapter-libsql (required for Prisma v7 "client" engine with SQLite at runtime — same pattern used in Phase 1 for auth).

```typescript
import { PrismaClient } from "@prisma/client";
import { PrismaLibSQL } from "@prisma/adapter-libsql";
import { createClient } from "@libsql/client";

const globalForPrisma = globalThis as unknown as {
  prisma: PrismaClient | undefined;
};

function createPrismaClient(): PrismaClient {
  const libsql = createClient({
    url: process.env.DATABASE_URL ?? "file:../data/manta.db",
  });
  const adapter = new PrismaLibSQL(libsql);
  return new PrismaClient({ adapter });
}

export const prisma = globalForPrisma.prisma ?? createPrismaClient();

if (process.env.NODE_ENV !== "production") {
  globalForPrisma.prisma = prisma;
}
```
