    - autoWidth: worksheet columns have width > 0 after autoWidth() call

    laporanKeuangan.ts — buildLaporanKeuanganSheet(wb, summary: FinanceSummary, periodLabel: string):
      ws.columns = [{header:'Metrik', key:'metrik', width:30}, {header:'Nilai', key:'nilai', width:20, style:{numFmt:RUPIAH_FMT}}]
      applyHeaderStyle(ws.getRow(1))
      addRow for each P&L line: Pendapatan, HPP, Laba Kotor, Diskon, Piutang, Terbayar, Laba Bersih
      No total row needed (already individual lines). autoWidth(ws).

    transaksi.ts — buildTransaksiSheet(wb, rows: SaleRow[], filters: TransaksiFilters):
      Filter rows by: dateFrom/dateTo (row.date), customer (row.customer), paymentStatus ('PAID'|'UNPAID' → row.status).
      ws.columns = [Tanggal, No Ref, Konsumen, Grand Total(Rp), HPP Total(Rp), Laba Kotor(Rp), Diskon(Rp), Status]
      applyHeaderStyle(ws.getRow(1)), add filtered rows, add bold TOTAL row with sums, autoWidth(ws).

    detailItem.ts — buildDetailItemSheet(wb, rows: SaleRow[], filters: DetailItemFilters):
      Filter rows by: dateFrom/dateTo, customer. Then flatten each row.items into individual detail rows.
      Columns: [Tanggal, No Ref, Konsumen, Produk, Qty, HPP Unit(Rp), Harga Unit(Rp), Subtotal(Rp)]
      applyHeaderStyle(ws.getRow(1)), add detail rows, add TOTAL row (sum Subtotal), autoWidth(ws).

    piutang.ts — buildPiutangSheet(wb, rows: CustomerRow[], filters: PiutangFilters):
      Filter rows by: customer (row.customer), paymentStatus ('unpaid' → only rows where totalPiutang > 0).
      Columns: [Konsumen, Total Transaksi, Grand Total(Rp), Terbayar(Rp), Piutang(Rp), Avg Diskon %]
      applyHeaderStyle(ws.getRow(1)), add rows, add TOTAL row, autoWidth(ws).

    stock.ts — buildStockSheet(wb, rows: StockItem[], filters: StockFilters):
      Filter rows by: category (row.category).
      Columns: [SKU, Nama, Kategori, HPP(Rp), Harga Jual(Rp), Stock, Margin/Unit(Rp), Margin %]
      applyHeaderStyle(ws.getRow(1)), add rows, no total row for stock. autoWidth(ws).

    marginProduk.ts — buildMarginProdukSheet(wb, rows: ProductMarginRow[], filters: MarginProdukFilters):
      Filter rows by: category (row.category).
      Columns: [SKU, Nama, Kategori, HPP Unit(Rp), Harga Jual(Rp), Harga Aktual(Rp), Margin %, Total Qty Sold]
      applyHeaderStyle(ws.getRow(1)), add rows, autoWidth(ws).

    marginKonsumen.ts — buildMarginKonsumenSheet(wb, rows: CustomerDiscountRow[]):
      No filter (no applicable filters in CONTEXT.md for this sheet).
      Columns: [Konsumen, Total Transaksi, Grand Total(Rp), Avg Diskon %]
      applyHeaderStyle(ws.getRow(1)), add rows, autoWidth(ws).

    Export async function buildWorkbook(config: ExportConfig, data: ExportData): Promise<ExcelJS.Workbook>
      Create new ExcelJS.Workbook()

        if config.sheets.includes(key): call the corresponding builder function with wb + relevant data + relevant filters
      return workbook
